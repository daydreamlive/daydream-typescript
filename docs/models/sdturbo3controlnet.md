# SDTurbo3Controlnet

## Example Usage

```typescript
import { SDTurbo3Controlnet } from "@daydreamlive/sdk/models";

let value: SDTurbo3Controlnet = {
  modelId: "daydreamlive/TemporalNet2-stable-diffusion-2-1",
  conditioningScale: 8815.16,
  preprocessor: "depth",
  enabled: false,
};
```
