# SDTurboControlnet

## Example Usage

```typescript
import { SDTurboControlnet } from "@daydreamlive/sdk/models";

let value: SDTurboControlnet = {
  modelId: "thibaud/controlnet-sd21-color-diffusers",
  conditioningScale: 7406.28,
  preprocessor: "upscale",
  enabled: true,
};
```
